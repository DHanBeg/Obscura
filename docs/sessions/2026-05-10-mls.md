## MLS API minimal surface
```rust
pub fn new_provider() -> OpenMlsRustCrypto;
pub struct Identity { did, credential_with_key, signature_keys }
impl Identity {
    fn generate(did, provider) -> Result<Self>;
}
pub fn generate_key_package(identity, provider) -> Result<KeyPackage>;
pub fn create_group(identity, provider) -> Result<MlsGroup>;
pub fn add_member(group, identity, kp, provider) -> Result<(Commit, Welcome)>;
pub fn process_welcome(welcome, provider) -> Result<MlsGroup>;
pub fn encrypt_message(group, identity, plaintext, provider) -> Result<MlsMessageOut>;
pub fn process_message(group, msg_in, provider) -> Result<Option<Vec<u8>>>;
```
